** Rooms Context Menu

**** Preface

Room entry context menus!
Room options are supplied dynamically by the active protocol engine via the
~:get-room-options~ bind so each protocol decides which actions apply to a
given room, space, server, channel, community, or direct message.

(defn- normalize-dispatch-event [ev]
  (if (and (sequential? ev) (seq ev))
    (let [v (vec ev)]
      (if (string? (first v))
        (assoc v 0 (keyword (first v)))
        v))
    ev))

(defn execute-option-command! [{:keys [action command]}]
  (cond
    (fn? action)
    (action)

    (map? command)
    (case (keyword (:type command))
      :dispatch
      (when-let [ev (normalize-dispatch-event (:event command))]
        (re-frame/dispatch ev))

      :dispatch-many
      (doseq [ev (:events command)]
        (when-let [norm (normalize-dispatch-event ev)]
          (re-frame/dispatch norm)))

      :engine-rpc
      (let [{:keys [handler arguments on-success-dispatch]} command]
        (go
          (try
            (let [res (<! (mesh/do-with-thread!
                           :engine-pool
                           {:handler   (keyword handler)
                            :arguments (or arguments {})}))]
              (when (and on-success-dispatch (ok-status? res))
                (re-frame/dispatch (normalize-dispatch-event on-success-dispatch))))
            (catch :default e
              (log/error "Room option RPC failed:" handler e)))))

      nil)

    :else nil))

(defn- options-cache-key [engine-id {:keys [room-id context membership is-forum?]}]
  [(or engine-id :default)
   (str room-id)
   (keyword (or context :list))
   (str (or membership "joined"))
   (boolean is-forum?)])

(defn fallback-room-actions [{:keys [room-id is-space? membership]}]
  (let [joined? (= (str/lower-case (str (or membership "joined"))) "joined")]
    (vec
     (remove nil?
             [(when joined?
                {:id        "settings"
                 :label-key :navigation.actions/settings
                 :icon      :settings
                 :command   {:type  :dispatch
                             :event [:rooms/open-settings room-id]}})
              (when joined?
                {:id         "leave"
                 :label-key  (if is-space?
                               :navigation.actions/leave-space
                               :navigation.actions/leave-room)
                 :icon       :leave
                 :class-name "danger"
                 :command    {:type  :dispatch
                              :event [:rooms/leave room-id]}})
              {:id        "copy-link"
               :label-key :navigation.actions/copy-link
               :icon      :link
               :command   {:type  :dispatch
                           :event [:rooms/copy-link room-id]}}]))))

(re-frame/reg-event-db
 :room-options/set
 (fn [db [_ cache-key options]]
   (assoc-in db [:room-options/by-key cache-key] (vec options))))

(re-frame/reg-sub
 :room-options/items
 (fn [db [_ cache-key]]
   (get-in db [:room-options/by-key cache-key])))

(re-frame/reg-event-fx
 :room-options/fetch
 (fn [{:keys [db]} [_ req]]
   (let [eng-id    (:active-engine db)
         cache-key (options-cache-key eng-id req)
         fallback! (fn [reason]
                     (log/warn "Engine :get-room-options fallback:" reason)
                     (re-frame/dispatch [:room-options/set cache-key (fallback-room-actions req)]))]
     (when (:room-id req)
       (go
         (try
           (let [res  (<! (mesh/do-with-thread!
                           :engine-pool
                           {:handler   :get-room-options
                            :arguments req}))
                 opts (when (ok-status? res)
                        (decode-edn-field res :options-edn :options))]
             (if (sequential? opts)
               (re-frame/dispatch [:room-options/set cache-key opts])
               (fallback! res)))
           (catch :default e
             (fallback! e))))))
   {}))

(defn ^:ui room-context-menu-content [{:keys [room-id parent-id room-name is-space? is-dm? context membership close-fn]}]
  (let [tr @(re-frame/subscribe [:i18n/tr])
        items (build-room-actions tr room-id parent-id room-name is-space? is-dm? context membership)]
    [:<>
     (for [{:keys [id label action class-name icon]} items]
       ^{:key id}
       [:div.context-menu-item
        {:class class-name
         :on-click (fn [e]
                     (.stopPropagation e)
                     (when action
                       (action))
                     (if close-fn
                       (close-fn)
                       (re-frame/dispatch [:ui/close-context-menu])))}
        (when icon [:span.item-icon icon])
        [:span.item-label label]])]))

(defmethod context-menu-component :room-actions [_]
  room-context-menu-content)
