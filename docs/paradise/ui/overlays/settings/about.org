(defn ^:ui about-tab []
  (let [tr       @(re-frame/subscribe [:i18n/tr])
        version  @(re-frame/subscribe [:app/version])
        app-name (:app-name config "Matrix Client")]
    [:div.settings-tab-content
     [:div.settings-hub-group
      [:div.settings-hub-group-label
       (tr [:settings.about/title])]
      [:div.settings-hub-group-card
       [:div.settings-card-stack
        [:div.settings-inline-header
         [:h3.settings-subheading app-name]
         [:div.settings-badge
          (str "v" version)]]
        [:p.settings-description
         (tr [:settings.about/description]
             [app-name])]]]]

     [:div.settings-hub-group
      [:div.settings-hub-group-label
       (tr [:settings.about/updates-title])]
      [:div.settings-hub-group-card
       [:div.settings-card-row
        [:div.settings-row-text
         [:div.settings-row-title
          (tr [:settings.about/updates-title])]
         [:div.settings-row-description
          (tr [:settings.about/updates-desc])]]
        [:button.settings-button.secondary
         {:on-click #(re-frame/dispatch [:app/poll-version true])}
         (tr [:settings.about/check-updates])]]]]]))
