**** Multi-method Modal

To avoid a nasty set of circular dependencies we actually want to centrally register all the components in one space.
This allows us to neatly create states and provide a single component to the main application to render.
#+begin_src clojurescript :tangle ../../../../src/paradise/ui/overlays/base.cljs
(defmulti modal-component identity)

(defmethod modal-component :default [modal-id]
  (fn [_props]
    [:div {:style {:padding "24px" :background "white" :color "black" :border-radius "8px"}}
     [:p modal-id]]))

(defmethod modal-component :plugin-portal [_]
  (fn [{:keys [render-fn] :as props}]
    [render-fn props]))

#+end_src

**** Multi-method Context Menu


#+begin_src clojurescript :tangle ../../../../src/paradise/ui/overlays/base.cljs
(defmulti context-menu-component identity)

(defmethod context-menu-component :default [menu-id]
  (fn [_props]
    [:div {:style {:padding "10px" :color "red"}}
     (str "Missing context menu: " menu-id)]))

(defmethod context-menu-component :plugin-portal [_]
  (fn [{:keys [render-fn] :as props}]
    [render-fn props]))

(defn ^:ui context-menu-items
  "Renders a list of `{:keys [id label dispatch action class-name icon]}` context menu entries."
  [items]
  [:<>
   (for [{:keys [id label dispatch action class-name icon]} items]
     ^{:key (or id label)}
     [:div.context-menu-item
      {:class    class-name
       :on-click (fn [e]
                   (.stopPropagation e)
                   (when dispatch
                     (re-frame/dispatch dispatch))
                   (when action
                     (action))
                   (re-frame/dispatch [:ui/close-context-menu]))}
      (when icon [:span.item-icon icon])
      [:span.item-label label]])])

(defn ^:ui feedback-banners
  "Renders standard error and/or success feedback banners inside overlay tabs."
  [{:keys [error success style]}]
  (let [attrs (when style {:style style})]
    [:<>
     (when error
       [:div.emotes-mgr-banner.is-error attrs error])
     (when success
       [:div.emotes-mgr-banner.is-success attrs success])]))
#+end_src
