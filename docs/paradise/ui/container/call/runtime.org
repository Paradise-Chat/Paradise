(defn- decode-base64-key
  [key-b64-string]

  (let [std-b64
        (-> key-b64-string
            (str/replace "-" "+")
            (str/replace "_" "/"))

        remainder
        (mod
         (.-length std-b64)
         4)

        padded-b64
        (if (zero? remainder)

          std-b64

          (str
           std-b64
           (str/join
            (repeat
             (- 4 remainder)
             "="))))

        raw-str
        (js/atob
         padded-b64)

        length
        (.-length raw-str)

        key-buffer
        (js/Uint8Array.
         length)]

    (dotimes [i length]
      (aset
       key-buffer
       i
       (.charCodeAt
        raw-str
        i)))

    key-buffer))





(defn replay-known-keys! []
  (when-let [kp @!e2ee-key-provider]
    (doseq [[participant-id keys-map] @!known-keys
            [key-index crypto-key] keys-map]
      (log/debug "Replaying LiveKit E2EE key" participant-id key-index)
      (.onSetEncryptionKey kp crypto-key participant-id (js/Number. key-index)))))

(defn inject-key!
  [participant-id key-index key-b64-string]
  (try
    (let [key-buffer (decode-base64-key key-b64-string)]
      (p/let [crypto-key (js/crypto.subtle.importKey
                          "raw"
                          (.-buffer key-buffer)
                          "HKDF"
                          false
                          #js ["deriveBits" "deriveKey"])]
        (log/debug "Injecting LiveKit E2EE key" participant-id key-index)
        (swap! !known-keys assoc-in [participant-id key-index] crypto-key)
        (when-let [kp @!e2ee-key-provider]
          (.onSetEncryptionKey kp crypto-key participant-id (js/Number. key-index)))
        crypto-key))
    (catch :default err
      (p/rejected err))))

(defn get-lk-options []
  (if-let [opts @!cached-lk-options]
    opts
    (let [kp   (new (.-BaseKeyProvider lkc) #js {:ratchetWindowSize 10 :keyringSize 256})
          opts #js {:e2ee #js {:keyProvider kp
                               :worker (js/Worker. "/livekit-e2ee-worker.js")}
                    :audioCaptureDefaults (microphone-capture-options)
                    :videoCaptureDefaults #js {:resolution (.-hd1080 (.-VideoPresets lkc))}
                    :publishDefaults      #js {:simulcast false :videoCodec "vp8"}}]

      (reset! !e2ee-key-provider kp)
      (reset! !cached-lk-options opts)
      (replay-known-keys!)
      opts)))

(defn- clamp01 [x]
  (max 0.0
       (min 1.0
            (or x 0.0))))


(defn- microphone-source []
  (.. lkc -Track -Source -Microphone))


(defn switch-device!
  [kind device-id]

  (if-let [room
           @!room]

    (.switchActiveDevice
     room
     kind
     device-id)

    (p/rejected
     (js/Error.
      "No active LiveKit room"))))


(defn get-or-create-audio-processor! []
  (or
   @!audio-processor

   (let [processor
         (audio-processor/create-processor
          {:threshold @!voice-threshold
           :gate-enabled? @!voice-gate-enabled?
           :hangover-ms 180})]

     (reset!
      !audio-processor
      processor)

     (when-let [set-enabled
                (aget processor
                      "setNoiseSuppressionEnabled")]
       (set-enabled
        @!noise-suppression-enabled?))

     processor)))

(defn- detach-local-track-published-listener!
  []
  (when-let [room
             @!room]

    (when-let [listener
               @!local-track-published-listener]

      (.off
       room
       (.-LocalTrackPublished
        (.-RoomEvent lkc))
       listener)))

  (reset!
   !local-track-published-listener
   nil))


(defn- attach-local-track-published-listener!
  [room]
  (let [listener
        (fn [publication _participant]
          (when (= (.-source publication)
                   (microphone-source))

            (when-let [track
                       (.-track publication)]

              (-> (ensure-microphone-processor!
                   track)

                  (p/catch
                   (fn [err]
                     (log/error
                      err
                      "Failed to attach Paradise processor to published microphone")))))))]

    (.on
     room
     (.-LocalTrackPublished
      (.-RoomEvent lkc))
     listener)

    (reset!
     !local-track-published-listener
     listener)

    listener))


(defn current-room []
  @!room)


(defn set-room!
  [room]
  (detach-local-track-published-listener!)

  (reset!
   !room
   room)

  (when room
    (attach-local-track-published-listener!
     room)

    (-> (ensure-active-microphone-processor!)

        (p/catch
         (fn [err]
           (log/error
            err
            "Failed to attach Paradise processor to existing microphone")))))

  room)


(defn clear-room! []
  (detach-local-track-published-listener!)

  (reset!
   !room
   nil))


(defn local-participant []

(defn ensure-microphone-processor!
  [track]

  (if-not track

    (p/resolved nil)

    (let [current
          (.getProcessor track)]

      (if current

        (do
          (reset!
           !audio-processor
           current)

          (reset!
           !audio-processor-track
           track)

          (apply-processor-settings!
           current)

          (p/resolved
           track))

        (let [processor
              (get-or-create-audio-processor!)]

          (-> (p/let [_ (.setProcessor
                         track
                         processor)]

                (reset!
                 !audio-processor
                 processor)

                (reset!
                 !audio-processor-track
                 track)

                (apply-processor-settings!
                 processor)

                (log/info
                 "Paradise audio processor attached to published microphone"
                 {:processor
                  (aget processor "name")

                  :track-id
                  (some-> track
                          .-mediaStreamTrack
                          .-id)})

                track)

              (p/catch
               (fn [err]
                 (when (identical?
                        processor
                        @!audio-processor)
                   (reset!
                    !audio-processor
                    nil))

                 (when (identical?
                        track
                        @!audio-processor-track)
                   (reset!
                    !audio-processor-track
                    nil))

                 (throw err)))))))))

(defn- apply-to-live-audio-processors!
  [f]
  (let [track-processor
        (current-microphone-processor)

        runtime-processor
        @!audio-processor]

    (when track-processor
      (f track-processor))

    (when (and runtime-processor
               (not (identical?
                     runtime-processor
                     track-processor)))
      (f runtime-processor))))

(defn set-microphone-enabled!

  (if-let [participant
           (local-participant)]

    (if enabled?

      (p/let
          [publication
           (.setMicrophoneEnabled
            participant
            true
            (microphone-capture-options))

           _
           (if-let [track
                    (some->
                     publication
                     .-track)]

             (ensure-microphone-processor!
              track)

             (ensure-active-microphone-processor!))]

        publication)

      (.setMicrophoneEnabled
       participant
       false))

    (p/rejected
     (js/Error.
      "No active LiveKit room"))))

(defn destroy-audio-processor!
  []
  (let [track
        @!audio-processor-track

        processor
        @!audio-processor]

    (if (and track
             (identical?
              processor
              (.getProcessor track)))

      (-> (.stopProcessor
           track
           false)

          (p/catch
           (fn [err]
             (log/debug
              err
              "Failed to stop LiveKit microphone processor"))))

      (when-let [destroy
                 (some-> processor
                         (aget "destroy"))]

        (-> (destroy)

            (p/catch
             (fn [err]
               (log/debug
                err
                "Audio processor cleanup failed"))))))

    (reset!
     !audio-processor-track
     nil)

    (reset!
     !audio-processor
     nil)))


(defn disconnect! []
  (if-let [room
           @!room]

    (.disconnect room)

    (p/resolved nil)))


(defn reset-session! []
  (destroy-audio-processor!)

  (clear-room!)

  (reset!
   !e2ee-key-provider
   nil)

  (reset!
   !cached-lk-options
   nil)

  (reset!
   !known-keys
   {}))
