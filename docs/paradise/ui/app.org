
* App


After having explored setting up and interfacing with the Rust sdk I believe we're in a good state to actually
start building out the *actual* UI.
Now re-frame more or less trivializes this and enables us to write a UI and then wire it up later.


**** Preface

For base functionality I would define the minimum usable product as:
- Spaces navigation (and DMs)
- Room navigation
- Timeline
- Message box
- Login

So these would be the first things to build on.
We'll continue to keep our minimalist login page, but we'll start trying to do a bit more polish on the
actual UI components themselves.

***** i18n
So we utilize a library written by the same individual who created the logging solution we use.
It is called Tempura. It provides a nice and easy way to actually create mappings to entries
and easy use around the repository to utilize them.

We want to be sure with the reframe components that we don't call unnecessarily. So segmenting
locale and dictionary allows us to only ever update strings if we explicitly change the
displayed language.
#+begin_src clojurescript :tangle ../../../src/paradise/ui/app.cljs
(re-frame/reg-sub
 :i18n/locale
 (fn [db _] (:locale db)))

(re-frame/reg-sub
 :i18n/dictionary
 (fn [db _] (:dictionary db)))

(re-frame/reg-sub
 :i18n/engine-dictionary
 (fn [db _] (:i18n/engine-dictionary db)))

(re-frame/reg-sub
 :i18n/plugin-dictionaries
 (fn [db _] (:i18n/plugin-dictionaries db {})))

(defonce !tr-cache (atom {:locale nil :is-empty? true :fn nil}))

(re-frame/reg-sub
 :i18n/tr
 :<- [:i18n/locale]
 :<- [:i18n/dictionary]
 :<- [:i18n/engine-dictionary]
 :<- [:i18n/plugin-dictionaries]
 (fn [[locale dictionary engine-dictionary plugin-dictionaries] _]
   (let [base-dict      (if (and (map? dictionary) (seq dictionary))
                          dictionary
                          i18n-data)
         plugin-dicts   (when (and (map? plugin-dictionaries) (seq plugin-dictionaries))
                          (map second (sort-by (comp str key) plugin-dictionaries)))
         effective-dict (if (or (and (map? engine-dictionary) (seq engine-dictionary))
                                (seq plugin-dicts))
                          (apply deep-merge base-dict engine-dictionary plugin-dicts)
                          base-dict)
         safe-locale    (or (some-> locale keyword) :en)]
     (if (empty? effective-dict)
       (fn [k & _] (str "[" (name (last k)) "]"))
       (partial tempura/tr {:dict effective-dict} [safe-locale :en])))))

(re-frame/reg-event-db
 :i18n/set-dictionary
 (fn [db [_ dict]]
   (assoc db :dictionary dict)))

(re-frame/reg-event-db
 :i18n/set-engine-dictionary
 (fn [db [_ engine-id dict]]
   (let [eng-kw  (some-> engine-id keyword)
         want-kw (:i18n/loading-engine-id db)]
     (if (or (nil? want-kw) (nil? eng-kw) (= want-kw eng-kw))
       (assoc db
              :i18n/engine-id eng-kw
              :i18n/engine-dictionary (when (map? dict) dict))
       db))))

(re-frame/reg-event-db
 :i18n/clear-engine-dictionary
 (fn [db _]
   (dissoc db :i18n/engine-id :i18n/engine-dictionary :i18n/loading-engine-id)))

(re-frame/reg-event-db
 :i18n/set-plugin-dictionary
 (fn [db [_ plugin-id dict]]
   (if-let [pid (some-> plugin-id keyword)]
     (if (and (map? dict) (seq dict))
       (assoc-in db [:i18n/plugin-dictionaries pid] dict)
       (update db :i18n/plugin-dictionaries dissoc pid))
     db)))

(re-frame/reg-event-db
 :i18n/remove-plugin-dictionary
 (fn [db [_ plugin-id]]
   (if-let [pid (some-> plugin-id keyword)]
     (update db :i18n/plugin-dictionaries dissoc pid)
     db)))

(defn- fetch-i18n-edn [url]
  (go
    (when (and (string? url) (not (str/blank? url)))
      (try
        (let [busted (str url
                          (if (str/includes? url "?") "&t=" "?t=")
                          (.getTime (js/Date.)))
              res    (<! (registry/fetch-edn busted))]
          (when (map? res) res))
        (catch :default _ nil)))))

(re-frame/reg-event-fx
 :i18n/load-engine-extensions
 (fn [{:keys [db]} [_ engine-id]]
   (if-let [eng-kw (some-> engine-id keyword)]
     (let [manifest-i18n-url (or (registry/get-engine-i18n-url eng-kw)
                                 (get-in db [:available-engines eng-kw :i18n-url])
                                 (get-in db [:available-engines eng-kw :i18n]))]
       (go
         (let [bind-dict (try
                           (let [res  (<! (mesh/do-with-thread!
                                           :engine-pool
                                           {:handler   (keyword (name eng-kw) "get-i18n")
                                            :arguments {}}))
                                 dict (when (contains? #{"success" :success} (:status res))
                                        (decode-edn-field res :dictionary-edn :dictionary))]
                             (when (map? dict) dict))
                           (catch :default _ nil))
               http-dict (<! (fetch-i18n-edn manifest-i18n-url))
               merged    (deep-merge bind-dict http-dict)]
           (re-frame/dispatch [:i18n/set-engine-dictionary eng-kw (or merged {})])))
       {:db (assoc db :i18n/loading-engine-id eng-kw)})
     {:dispatch [:i18n/clear-engine-dictionary]})))

(re-frame/reg-event-fx
 :i18n/load-plugin-extensions
 (fn [_ [_ plugin-id source-url plugin-map]]
   (when-let [pid (some-> plugin-id keyword)]
     (let [inline-dict (when (map? (:i18n plugin-map))
                         (:i18n plugin-map))
           i18n-url    (or (registry/resolve-manifest-i18n-url plugin-map source-url)
                           (registry/get-plugin-i18n-url pid))]
       (cond
         (and inline-dict (not i18n-url))
         {:dispatch [:i18n/set-plugin-dictionary pid inline-dict]}

         i18n-url
         (do
           (go
             (let [http-dict (<! (fetch-i18n-edn i18n-url))
                   merged    (deep-merge inline-dict http-dict)]
               (when (seq merged)
                 (re-frame/dispatch [:i18n/set-plugin-dictionary pid merged]))))
           {})

         :else
         {})))))

(re-frame/reg-event-db
 :i18n/set-locale
 (fn [db [_ locale]]
   (assoc db :locale locale)))

(re-frame/reg-fx
 :ui/hotswap-css
 (fn [new-href]
   (if-let [link (.getElementById js/document "app-theme")]
     (do
       (set! (.-href link) new-href)
       (log/info "Swapped CSS to:" new-href))
     (log/error "Theme stylesheet link #app-theme not found!"))))

(re-frame/reg-event-fx
 :ui/switch-theme
 (fn [{:keys [db]} [_ theme-val]]
   (let [safe-theme (or theme-val "default")
         theme-str  (if (keyword? safe-theme) (name safe-theme) safe-theme)
         is-url?    (str/starts-with? theme-str "http")
         href       (if is-url?
                      theme-str
                      (str "css/" (case (keyword theme-str)
                                    :discord "cordlike.css"
                                    :matrix  "matrix.css"
                                    :retro   "retro.css"
                                    "app.css")))]
     {:db             (assoc db :ui/current-theme theme-str)
      :dispatch       [:settings/save "theme" theme-str]
      :ui/hotswap-css href})))

**** Mounting the Application

This is to mount our application at the element with the ID "root" inside
our index.html. It's effectively just an attachment point for us to latch
onto.
Also inside is th init function which informs reframe to start up
as well as a method that instructs shadow-cljs how to handle the re-render
when we detatch and reload (basically when you hit control + s to save).
#+begin_src clojurescript :tangle ../../../src/paradise/ui/app.cljs
(defonce root (atom nil))

(defn mount-root []
  (when (and (exists? js/document) (exists? js/Capacitor))
    (when-let [doc-el (.-documentElement js/document)]
      (.setAttribute doc-el "data-platform" (.getPlatform js/Capacitor))))
  (re-frame/dispatch [:ui/switch-theme])
  (re-frame/clear-subscription-cache!)
  (let [container (.getElementById js/document "root")]
    (when-not @root
      (reset! root (rdom/create-root container)))
    (init-window-size-listener!)
    (init-capacitor-listeners!)
    (.render @root (r/as-element [main-layout]))))

(defn ^:export init []
  (re-frame/dispatch-sync [:initialize-db])
  (notifications/init-audio-unlock!)
  (call/init-native-listeners!)
  (tray/init-tray!)
  (re-frame/dispatch [:app/load-settings-by-stage :boot])
  (re-frame/dispatch [:app/start-boot-sequence])
  (mount-root))

(re-frame/reg-event-fx
 :app/start-boot-sequence
 (fn [_ _]
   (-> (load-config)
       (p/then (fn [config]
                 (reset! !config config)
                 (re-frame/dispatch-sync [:app/config-loaded config])
                 (log/info "Config loaded:" config)
                 (load-i18n)))
       (p/then (fn [_]
                 (let [params  (js/URLSearchParams. js/window.location.search)
                       room-id (.get params "room")]
                   (when room-id
                     (log/info "Deep link detected on boot:" room-id)
                     (re-frame/dispatch [:nav/lock-deep-link room-id])
                     (.replaceState js/window.history #js {} "" (.. js/window -location -pathname)))

                   (p/let [saved-user (store/get-setting "last_active_user")]
                     (re-frame/dispatch [:app/cold-boot saved-user]))

                   (notifications/init!)
                   (js/setInterval #(re-frame/dispatch [:app/poll-version false]) (* 1000 60 15)))))
       (p/catch #(log/error "App initialization failed:" %)))
   {}))


(re-frame/reg-event-db
 :nav/lock-deep-link
 (fn [db [_ room-id]]
   (assoc db :deep-link-room room-id)))

(re-frame/reg-event-fx
 :nav/route-from-push
 (fn [{:keys [db]} [_ room-id]]
   (if (= (:auth-status db) :logged-in)
     {:dispatch [:rooms/select room-id]}
     {:dispatch [:nav/lock-deep-link room-id]})))

(re-frame/reg-event-fx
 :app/restore-user-session
 (fn [{:keys [db]} [_ user-id]]
   (let [deep-room      (:deep-link-room db)
         pending-answer (get-in db [:call :pending-native-answer])]
     (if deep-room
       {:dispatch-n (cond-> [[:auth/set-status :logged-in]
                             [:rooms/select deep-room]
                             [:sdk/ignite-session]]
                      pending-answer
                      (conj [:call/flush-pending-native-answer]))}
       (do
         (p/let [[room-id space-id] (p/all [(store/get-setting (str "last_room_" user-id))
                                            (store/get-setting (str "last_space_" user-id))])]
           (if room-id
             (do
               (log/info "Restoring session for" user-id "-> Space:" space-id "| Room:" room-id)
               (when space-id
                 (re-frame/dispatch [:space/select space-id]))
               (re-frame/dispatch [:auth/set-status :logged-in])
               (re-frame/dispatch [:rooms/select room-id])
               (re-frame/dispatch [:sdk/ignite-session])
               (when pending-answer
                 (re-frame/dispatch [:call/flush-pending-native-answer])))
             (do
               (re-frame/dispatch [:auth/set-status :logged-in])
               (re-frame/dispatch [:sdk/ignite-session])
               (when pending-answer
                 (re-frame/dispatch [:call/flush-pending-native-answer])))))
         {})))))


(defn ^:after-load re-render []
  (load-i18n)
  (mount-root))
