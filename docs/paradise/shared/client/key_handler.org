
** Key Handler


**** Preface

key listener.

Later, we'll expand it to handle native input for features like push to talk.

**** Multi-method

This allows for our namespace here to now be ignorant as to what key binds
are registered. Instead that responsibility is pushed to the actual
module namespaces themselves.

#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/key_handler.cljs
(def storage-key "paradise.settings.keybinds")

(defn- normalize-id [id]
  (if (keyword? id)
    (subs (str id) 1)
    (str id)))

(defmulti keybind-spec identity)

(defmethod keybind-spec :default [action-id]
  nil)

(defonce !registry (r/atom {}))

(defonce !last-fired (atom {}))

(defn all-keybind-specs []
  (let [mm-entries (reduce-kv
                    (fn [acc dispatch-val method-fn]
                      (if (= dispatch-val :default)
                        acc
                        (if-let [spec (method-fn dispatch-val)]
                          (let [norm-id (normalize-id (or (:id spec) dispatch-val))]
                            (assoc acc norm-id (merge {:global? false :allow-in-input? true}
                                                      spec
                                                      {:id norm-id})))
                          acc)))
                    {}
                    (methods keybind-spec))]
    (merge mm-entries @!registry)))
#+end_src


**** Global Key Listener


#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/key_handler.cljs
(defui global-key-listener []
  (r/with-let [handler (fn [e]
                         (when (and (= (.-key e) "k")
                                    (or (.-metaKey e) (.-ctrlKey e)))
                           (.preventDefault e)
                           (re-frame/dispatch [:ui/open-modal :quick-switcher
                                               {:backdrop-props {:class "lightbox-backdrop"}
                                                :window-props   {:style {:background "transparent"
                                                                      :box-shadow "none"}}}])))]
    (.addEventListener js/window "keydown" handler)
    (finally
      (.removeEventListener js/window "keydown" handler))))
#+end_src
