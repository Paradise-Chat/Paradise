**** Persistence Helpers

We need standard low-level helpers to read and write raw JSON to the OPFS.
We keep these as simple JavaScript object operations to avoid unnecessary overhead from
converting back and forth to Clojure maps constantly.
#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn- load-raw-sessions-opfs
  ([engine-id]
   (p/let [dir (.. js/navigator -storage getDirectory)]
     (load-raw-sessions-opfs dir engine-id true)))
  ([dir engine-id create?]
   (-> (p/let [file-handle (.getFileHandle dir
                                           (engine-file-name engine-id)
                                           #js {:create (boolean create?)})
               sync-handle (.createSyncAccessHandle file-handle)]
         (try
           (let [size (.getSize sync-handle)]
             (if (zero? size)
               #js {}
               (let [buffer (js/Uint8Array. size)
                     _      (.read sync-handle buffer #js {:at 0})
                     text   (.decode (js/TextDecoder.) buffer)]
                 (if (str/blank? text)
                   #js {}
                   (js/JSON.parse text)))))
           (finally
             (.close sync-handle))))
       (p/catch (fn [_] #js {})))))

(defn- save-raw-sessions-opfs! [engine-id sessions-obj]
  (p/let [dir (.. js/navigator -storage getDirectory)
          file-handle (.getFileHandle dir (engine-file-name engine-id) #js {:create true})
          sync-handle (.createSyncAccessHandle file-handle)]
    (try
      (let [encoder (js/TextEncoder.)
            text (.stringify js/JSON sessions-obj)
            buffer (.encode encoder text)]
        (.truncate sync-handle 0)
        (.write sync-handle buffer #js {:at 0})
        (.flush sync-handle))
      (finally
        (.close sync-handle)))))
#+end_src




**** Engine Helpers

We need a neat manner to load from the different engines we might encounter.

#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn- format-stored-account-label [uid username display-name]
  (let [u (some-> username str str/trim not-empty)
        d (some-> display-name str str/trim not-empty)]
    (cond
      (and d u (not= (str/lower-case d) (str/lower-case u)))
      (str d " (@" u ")")

      u
      (str "@" u)

      d
      d

      :else
      (str uid))))

(defn- discover-opfs-session-engines [dir seed-engines]
  (let [seed (->> seed-engines
                  (map name)
                  (remove #(or (str/blank? %) (= % "default")))
                  vec)]
    (if (and dir (fn? (.-entries dir)))
      (let [iter (.entries dir)]
        (p/loop [acc seed]
          (p/let [step (-> (.next iter) (p/catch (fn [_] #js {:done true})))]
            (if (.-done step)
              (vec (distinct acc))
              (let [val   (.-value step)
                    fname (when val (aget val 0))
                    m     (when (string? fname)
                            (re-matches #"^sessions_(.+)\.json$" fname))
                    eng   (second m)]
                (p/recur (if (and eng (not= eng "default"))
                           (conj acc eng)
                           acc)))))))
      (p/resolved (vec (distinct seed))))))

(defn- read-opfs-engine-accounts [dir eng-str]
  (p/let [parsed (load-raw-sessions-opfs dir eng-str false)
          uids   (if (and parsed (object? parsed))
                   (js/Object.keys parsed)
                   #js [])]
    (vec
     (keep (fn [uid]
             (when-let [entry (gobj/get parsed uid)]
               (let [raw (or (gobj/get entry "session") entry)]
                 (when (and raw (object? raw))
                   (let [tok   (or (gobj/get raw "accessToken")
                                   (gobj/get raw "token"))
                         url   (or (gobj/get entry "serverUrl")
                                   (gobj/get raw "serverUrl")
                                   (gobj/get raw "apiBase")
                                   "")
                         uname (gobj/get raw "username")
                         dname (or (gobj/get raw "displayName")
                                   (gobj/get raw "display_name"))
                         av    (or (gobj/get raw "avatarUrl")
                                   (gobj/get raw "avatar_url"))]
                     (when (or tok (not (str/blank? (str url))) (gobj/get raw "userId"))
                       {:userId       (str uid)
                        :user-id      (str uid)
                        :engine       eng-str
                        :label        (format-stored-account-label uid uname dname)
                        :username     uname
                        :display-name dname
                        :avatar       av
                        :avatar-url   av
                        :server-url   url
                        :base-url     url
                        :server       url}))))))
           uids))))

(defn list-all-stored-accounts [seed-engines]
  (p/let [dir     (.. js/navigator -storage getDirectory)
          eng-ids (discover-opfs-session-engines dir seed-engines)]
    (p/loop [remaining eng-ids
             out       []]
      (if (empty? remaining)
        out
        (p/let [batch (read-opfs-engine-accounts dir (first remaining))]
          (p/recur (rest remaining) (into out batch)))))))

We also need a cryptographic implementation for generating passphrases.
The Rust SDK uses this passphrase to encrypt the IndexedDB where the actual keys and messages live.
#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn generate-passphrase []
  (let [array (js/Uint8Array. 32)]
    (.getRandomValues js/crypto array)
    (js/btoa (.apply js/String.fromCharCode nil array))))
#+end_src


#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn generate-uuid []
  (if (and (exists? js/crypto) (exists? (.-randomUUID js/crypto)))
    (.randomUUID js/crypto)
    (let [template "xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx"]
      (.replace template
                (js/RegExp. "[xy]" "g")
                (fn [c]
                  (let [r (bit-or (* (js/Math.random) 16) 0)
                        v (if (= c "x") r (bit-or (bit-and r 0x3) 0x8))]
                    (.toString v 16)))))))
#+end_src


****** Settings IndexedDB

 We will need to persist settings for the applications usage.
Webviews or browsers, IndexedDB will work just fine for this.
OPFS is unideal for the main thread to use and we don't really
*need* it.

#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(def ^:private settings-db-name "paradise-settings")
(def ^:private settings-store-name "settings")

(defn- open-settings-db []
  (idb/open-db
   settings-db-name
   1
   (fn [db _]
     (when-not (.contains (.-objectStoreNames db) settings-store-name)
       (.createObjectStore db settings-store-name)))))

(defn- parse-stored-setting [key raw-val]
  (cond
    (nil? raw-val) nil
    (string? raw-val)
    (let [trimmed    (str/trim raw-val)
          first-char (when (pos? (count trimmed)) (subs trimmed 0 1))]
      (cond
        (= trimmed "true")  true
        (= trimmed "false") false
        (= trimmed "nil")   nil

        (re-matches #"^-?\d+(?:\.\d+)?(?:[eE][+-]?\d+)?$" trimmed)
        (try
          (reader/read-string trimmed)
          (catch :default _ raw-val))

        (#{"{" "[" "(" "#" ":" "\""} first-char)
        (try
          (if (str/starts-with? trimmed "#js ")
            (reader/read-string (str/replace trimmed #"#js\s+(?=[\[\{])" ""))
            (reader/read-string raw-val))
          (catch :default e
            (try
              (if (str/includes? raw-val "#js ")
                (reader/read-string (str/replace raw-val #"#js\s+(?=[\[\{])" ""))
                (do
                  (js/console.error "EDN Parse Error for key:" key "val:" raw-val e)
                  raw-val))
              (catch :default _
                (js/console.error "EDN Parse Error for key:" key "val:" raw-val e)
                raw-val))))

        :else raw-val))
    :else
    (js->clj raw-val :keywordize-keys true)))

(defn- normalize-setting-for-storage [value]
  (cond
    (array? value)
    (js->clj value :keywordize-keys true)

    (and (object? value) (= (type value) js/Object))
    (js->clj value :keywordize-keys true)

    :else
    value))

(defn get-setting [key]
  (p/let [raw (idb/get-record (open-settings-db) settings-store-name key)]
    (parse-stored-setting key raw)))

(defn set-setting! [key value]
  (if (some? value)
    (idb/put-record! (open-settings-db)
                     settings-store-name
                     (pr-str (normalize-setting-for-storage value))
                     key)
    (idb/delete-record! (open-settings-db) settings-store-name key)))

(defn nav-state-key [engine-id user-id]
  (let [eng (when engine-id (name engine-id))
        uid (some-> user-id str not-empty)]
    (when (and eng uid)
      (str "last_nav_" eng "_" uid))))

(defn save-nav-state! [engine-id user-id {:keys [space-id room-id filter-id]}]
  (when-let [k (nav-state-key engine-id user-id)]
    (let [payload {:space-id  (some-> space-id str not-empty)
                   :room-id   (some-> room-id str not-empty)
                   :filter-id (or (some-> filter-id str not-empty) "people")}]
      (set-setting! k payload))))

(defn load-nav-state [engine-id user-id]
  (let [k   (nav-state-key engine-id user-id)
        uid (some-> user-id str not-empty)]
    (p/let [atomic-val (when k (get-setting k))]
      (if (map? atomic-val)
        {:space-id  (some-> (:space-id atomic-val) str not-empty)
         :room-id   (some-> (:room-id atomic-val) str not-empty)
         :filter-id (or (some-> (:filter-id atomic-val) str not-empty) "people")}
        (if uid
          (p/let [[legacy-room legacy-space]
                  (p/all [(get-setting (str "last_room_" uid))
                          (get-setting (str "last_space_" uid))])]
            {:space-id  (some-> legacy-space str not-empty)
             :room-id   (some-> legacy-room str not-empty)
             :filter-id "people"})
          {:space-id nil :room-id nil :filter-id "people"})))))

**** IndexedDB Management

This function handles the deletion of the encrypted database.
It wraps the raw IDBRequest in a Promise so we can await its
completion before clearing the session metadata.
We standardize on the name prefix closura-store-.
#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn- delete-dbs! [db-names]
  (idb/delete-dbs! db-names))
#+end_src


There is a need to provide the token to the service workers if we're to
provide rich message data to the user. Using the IndexedDB is convenient
here as we can provide multi-account notification support without much
overhead by baking the preparation into the session handling.
It should be noted that the vulnerability is the same here as it would be
in the local storage. Any cross site scripting attack will impact either,
so the choice to do this isn't inherently worse.


It actually turned out multi-account notification support was limited
by Sygnal and the way Matrix sends out push notifs. In a goal to limit
the amount of identifiable information passed through.

Still, multi-protocol notification support is still quite feasible.


**** Implementation Logic

These functions form the "brain" of the store.
They are defined outside the deftype for easier testing and cleaner syntax.

Updates a specific user's session while preserving others.
#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn- save-session-impl! [engine-id account-id payload owned-dbs]
  (p/let [sessions (load-raw-sessions-opfs engine-id)
          existing (aget sessions account-id)
          merged   (js/Object.assign #js {} existing payload)]
    (when (seq owned-dbs)
      (aset merged "ownedDbs" (clj->js owned-dbs)))
    (aset sessions account-id merged)
    (save-raw-sessions-opfs! engine-id sessions)))
#+end_src

Removes the specific user's database and metadata.
#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(defn- clear-session-impl! [engine-id account-id]
  (p/let [sessions  (load-raw-sessions-opfs engine-id)
          data      (aget sessions account-id)
          owned-dbs (when data (js->clj (aget data "ownedDbs")))
          _         (delete-dbs! owned-dbs)]
    (js-delete sessions account-id)
    (save-raw-sessions-opfs! engine-id sessions)))
#+end_src


**** The SessionStore Class

Finally, we wrap everything in a deftype. This compiles to a JavaScript class,
initially this was to be simply compatible with the patterns expected by
the Matrix SDK and UI layers.
However, it is still useful to keep this pattern going forward for
our generic session storing.

We use Object to implement the methods directly on the prototype.
#+begin_src clojurescript :tangle ../../../../src/paradise/shared/client/session_store.cljs
(deftype SessionStore [engine-id]
  Object
  (loadSessions [this]
    (load-raw-sessions-opfs engine-id))

  (save [this account-id payload owned-dbs]
    (save-session-impl! engine-id account-id payload owned-dbs))

  (generatePassphrase [this]
    (generate-passphrase))

  (generateStoreId [this]
    (generate-uuid))

  (getStoreName [this store-id]
    (str "paradise-store-" (name engine-id) "-" store-id))

  (clear [this account-id]
    (clear-session-impl! engine-id account-id)))
#+end_src
