
** SCI Runner UI Common

**** Preface

Shared SCI namespace table and plugin-scoped macro/registry builders used by
both the main-thread UI SCI runner (=paradise.shared.sci-runner.ui=) and the
virtualizer worker SCI runner (=paradise.shared.sci-runner.virtualizer=).
Keeping this separate from =sci-runner.shared= ensures the engine worker does
not pull in UI namespaces.

(defn- make-log-macro [sym target-sym !current-eval-plugin]
  (sci/new-macro-var
   sym
   (fn [form _env & args]
     (list* target-sym @!current-eval-plugin (:line (meta form)) args))))

(defn make-timbre-namespace
  "Creates a plugin-attributed `taoensso.timbre` SCI namespace bound to `!current-eval-plugin`."
  [!current-eval-plugin]
  {'info         (make-log-macro 'info 'taoensso.timbre/plugin-info !current-eval-plugin)
   'warn         (make-log-macro 'warn 'taoensso.timbre/plugin-warn !current-eval-plugin)
   'error        (make-log-macro 'error 'taoensso.timbre/plugin-error !current-eval-plugin)
   'debug        (make-log-macro 'debug 'taoensso.timbre/plugin-debug !current-eval-plugin)
   'plugin-info  paradise.shared.utils.logger/plugin-info
   'plugin-warn  paradise.shared.utils.logger/plugin-warn
   'plugin-error paradise.shared.utils.logger/plugin-error
   'plugin-debug paradise.shared.utils.logger/plugin-debug})

(defn make-registry-namespace
  "Creates the `paradise.shared.client.registry` SCI namespace with plugin-tagged slot registration."
  [!current-eval-plugin]
  {'!components              registry/!components
   'reg-slot-item            (fn [slot-id item]
                               (registry/reg-slot-item
                                slot-id
                                (assoc item :plugin-id @!current-eval-plugin)))
   '!active-overrides        registry/!active-overrides
   '!height-estimators       registry/!height-estimators
   'reg-height-estimator!    (fn [id estimator-fn]
                               (registry/reg-height-estimator!
                                id estimator-fn @!current-eval-plugin))
   'remove-plugin-overrides! registry/remove-plugin-overrides!
   'get-slot                 registry/get-slot})

(defn build-ui-namespaces
  "Merges shared UI namespaces with plugin-bound logger/registry and runner-specific extras."
  [!current-eval-plugin extra-namespaces]
  (merge base-ui-namespaces
         {'taoensso.timbre                 (make-timbre-namespace !current-eval-plugin)
          'paradise.shared.client.registry (make-registry-namespace !current-eval-plugin)}
         extra-namespaces))
