(defn clean-shortcode [raw]
  (-> (if (keyword? raw) (name raw) (str raw))
      (str/replace #":" "")
      str/trim))

(defn slugify-shortcode
  "Normalizes an arbitrary string or filename into a valid emote shortcode slug.
   Options:
   - `:strip-ext?`  strip trailing file extension before slugifying (default false)
   - `:lowercase?`  force lower-case output (default false)
   - `:allow-plus?` keep `+` characters (default false)
   - `:min-len`     pad short slugs when provided
   - `:max-len`     truncate long slugs when provided"
  ([raw]
   (slugify-shortcode raw nil))
  ([raw {:keys [strip-ext? lowercase? allow-plus? min-len max-len]}]
   (let [base     (clean-shortcode (or raw ""))
         no-ext   (if strip-ext? (str/replace base #"\.[^.]+$" "") base)
         cased    (if lowercase? (str/lower-case no-ext) no-ext)
         pat      (if allow-plus? #"[^A-Za-z0-9_+-]+" #"[^A-Za-z0-9_-]+")
         slug     (-> cased
                      (str/replace pat "_")
                      (str/replace #"_+" "_")
                      (str/replace #"^_+|_+$" ""))
         bounded  (if (and max-len (> (count slug) max-len))
                    (subs slug 0 max-len)
                    slug)]
     (cond
       (empty? bounded) ""
       (and min-len (< (count bounded) min-len)) (str bounded "_emote")
       :else bounded))))

(defn format-shortcode [raw]
  (let [cleaned (clean-shortcode raw)]
    (if (seq cleaned)
      (str ":" cleaned ":")
      ":emote:")))

(defn extract-first-url
  "Extracts the first HTTP(S) URL from `text`, stripping trailing punctuation."
  [text]
  (when text
    (when-let [match (re-find #"https?://[^\s\"'<>]+" text)]
      (str/replace match #"[.,:;!?]$" ""))))

(defn- parse-style-str [style-str]
  (if (string? style-str)
    (->> (str/split style-str #";")
         (remove str/blank?)
         (map #(str/split % #":"))
         (filter #(= 2 (count %)))
         (map (fn [[k v]] [(keyword (str/trim k)) (str/trim v)]))
         (into {}))
    style-str))

(defn- transform-font-span [attrs]
  (let [bg             (or (get attrs :data-bg-color) (get attrs :data-mx-bg-color))
        fg             (or (get attrs :data-color) (get attrs :data-mx-color))
        existing-style (parse-style-str (get attrs :style))
        style-map      (cond-> (or existing-style {})
                         bg (assoc :background-color bg)
                         fg (assoc :color fg))]
    (if (seq style-map)
      (assoc attrs :style style-map)
      attrs)))

(defn- transform-a [attrs]
  (assoc attrs :rel "noopener" :target "_blank"))

(defn- valid-url? [href]
  (if (string? href)
    (let [scheme (-> href (str/split #":" 2) first str/lower-case)]
      (contains? url-schemes scheme))
    false))


(defn- transform-img [attrs]
  (let [src    (or (get attrs :data-emote-uri) (get attrs :src) "")
        alt    (get attrs :alt)
        emote? (or (contains? attrs :data-emote)
                   (contains? attrs :data-emote-id)
                   (contains? attrs :data-mx-emoticon)
                   (and (media-uri? src) (not (valid-url? src))))]
    (if (and (media-uri? src) (or emote? (valid-url? src)))
      {:tag   "comp:paradise.media.component/media-native"
       :attrs {:uri       src
               :emote-id  (get attrs :data-emote-id)
               :class     "timeline-emotes"
               :className "timeline-emotes"
               :alt       alt}}
      {:tag     :a
       :attrs   {:href src :rel "noopener" :target "_blank"}
       :content [(or alt src)]})))

(defn- filter-code-classes [attrs]
  (if-let [cls (:class attrs)]
    (let [classes (str/split cls #"\s+")
          valid   (filter #(str/starts-with? % "language-") classes)]
      (if (seq valid)
        (assoc attrs :class (str/join " " valid))
        (dissoc attrs :class)))
    attrs))

(defn sanitize-nodes [nodes depth]
  (if (> depth max-tag-nesting)
    []
    (into []
          (mapcat
           (fn [node]
             (cond
               (string? node)
               [node]

               (and (map? node) (= (:type node) :element))
               (let [{:keys [tag attrs content]} node]
                 (cond
                   (contains? non-text-tags tag)
                   []

                   (contains? permitted-tags tag)
                   (let [allowed-keys (get permitted-attrs tag #{})
                         clean-attrs  (select-keys attrs allowed-keys)
                         [final-tag final-attrs transformed-content]
                         (case tag
                           (:font :span) [tag (transform-font-span clean-attrs) nil]
                           :a            [tag (transform-a clean-attrs) nil]
                           :code         [tag (filter-code-classes clean-attrs) nil]
                           :img          (let [{t :tag a :attrs c :content} (transform-img clean-attrs)]
                                           [t a c])
                           [tag clean-attrs nil])
                         final-attrs  (if (and (= final-tag :a)
                                               (not (valid-url? (:href final-attrs))))
                                        (dissoc final-attrs :href)
                                        final-attrs)
                         children     (or transformed-content
                                          (sanitize-nodes content (inc depth)))]
                     [(if (seq final-attrs)
                        (into [final-tag final-attrs] children)
                        (into [final-tag] children))])

                   :else
                   (sanitize-nodes content depth)))

               :else []))
           nodes))))

(defn sanitize-custom-html [raw-html]
  (when raw-html
    (let [html-str      (str raw-html)
          raw-fragments (h/parse-fragment html-str)
          hickory-maps  (map h/as-hickory raw-fragments)]
      (sanitize-nodes hickory-maps 0))))

(defonce ^:private !sanitize-cache
  (mem-cache/make-ttl-cache {:max-entries 300}))

(defn memoized-sanitize
  "Bounded-cache wrapper around `sanitize-custom-html` for hot timeline paths."
  [html]
  (mem-cache/memoize-bounded !sanitize-cache html #(sanitize-custom-html html)))

(defn sanitize-text [raw-text]
  (when raw-text
    (str/escape (str raw-text) {\& "&amp;" \< "&lt;" \> "&gt;" \" "&quot;" \' "&#39;"})))

(defn linkify-text [text]
  (if (str/blank? text)
    (if (seq text) (list text) '())
    (let [pattern #"https?://[^\s]+"
          parts   (str/split text pattern -1)
          matches (re-seq pattern text)]
      (if (empty? matches)
        (list text)
        (mapcat
         (fn [i part]
           (let [url (nth matches i nil)]
             (if url
               (cond-> []
                 (seq part) (conj part)
                 true       (conj ^{:key (str "lnk-" i)}
                                  [:a {:href   url
                                       :target "_blank"
                                       :rel    "noopener noreferrer"}
                                   url]))
               (when (seq part) [part]))))
         (range (count parts))
         parts)))))

(def block-level-tags #{:p :div :h1 :h2 :h3 :h4 :h5 :h6 :blockquote :li :tr})

(defn hiccup->text [node]
  (cond
    (string? node) node
    (vector? node)
    (let [tag        (first node)
          has-attrs? (map? (second node))
          attrs      (if has-attrs? (second node) {})
          children   (if has-attrs? (drop 2 node) (drop 1 node))
          inner-text (str/join "" (map hiccup->text children))]
      (cond
        (= tag :br) "\n"
        (contains? block-level-tags tag) (str inner-text "\n")
        (or (:uri attrs) (= tag :img) (= tag "media-native") (= tag "comp:paradise.media.component/media-native"))
        (str " " (or (:alt attrs) "[e]") " ")
        :else inner-text))
    (sequential? node) (str/join "" (map hiccup->text node))
    :else ""))

(defn process-raw-event [e source room-id]
  (let [html-txt     (or (get-in e [:content :inner :content :html]) "")
        cleaned-html (if (seq html-txt)
                       (str/replace html-txt #"(?i)<br[^>]*>\s*(?=</(?:p|div|h[1-6]|blockquote|li)>)" "")
                       "")
        safe-hiccup  (when (seq cleaned-html)
                       (sanitize-custom-html cleaned-html))
        plain-text   (if safe-hiccup
                       (hiccup->text safe-hiccup)
                       (or (:body e)
                           (get-in e [:content :body])
                           (get-in e [:content :inner :content :body])
                           (get-in e [:content :caption])
                           (get-in e [:content :inner :content :caption])
                           ""))]
    (-> e
        (assoc :timeline-source source)
        (assoc :room-id room-id)
        (assoc :clean-hiccup safe-hiccup)
        (assoc :clean-text plain-text)
        (update :type keyword)
        (assoc :raw e))))

(defonce !file-registry (atom {}))

(defn register-file! [file]
  (let [id (str (random-uuid))]
    (log/info "Registering native File in memory registry. ID:" id "Name:" (.-name file))
    (swap! !file-registry assoc id file)
    id))

(defn get-file [id]
  (let [f (get @!file-registry id)]
    (log/info "Fetching file from registry. ID:" id "Found? ->" (some? f))
    f))

(defn unregister-file! [id]
  (log/info "Unregistering file from memory registry. ID:" id)
  (swap! !file-registry dissoc id))

(defn file->array-buffer [f]
  (p/create
   (fn [resolve-fn reject]
     (try
       (if-not f
         (resolve-fn (js/ArrayBuffer. 0))
         (if (.-arrayBuffer f)
           (-> (.arrayBuffer f) (.then resolve-fn) (.catch reject))
           (let [reader (js/FileReader.)]
             (set! (.-onload reader) #(resolve-fn (.-result reader)))
             (set! (.-onerror reader) #(reject (.-error reader)))
             (.readAsArrayBuffer reader f))))
       (catch :default e
         (reject e))))))

(defn file->base64-payload
  "Reads a browser File/Blob into `{:buffer-b64 :mime :filename}` for worker RPCs,
  or resolves to nil when `file` is nil."
  [^js file]
  (if-not file
    (p/resolved nil)
    (p/let [ab (file->array-buffer file)]
      (let [u8  (when (and ab (pos? (.-byteLength ab)))
                  (js/Uint8Array. ab))
            b64 (when u8 (encoding/uint8->base64 u8))]
        {:buffer-b64 b64
         :mime       (or (not-empty (.-type file)) "image/png")
         :filename   (.-name file)}))))

***** Metadata Probing

Only images receive a persistent preview ~blob:~ URL because videos, audio, and
generic files render icon placeholders. On iOS WKWebView, detached ~<video>~
probing can crash CoreMedia/GPUProcess and decoding full-resolution 12-48 MP
camera photos through ~new Image()~ spikes WebContent/GPU memory. Therefore,
~extract-metadata~ parses image dimensions directly from the first 64 KB header
slice (PNG, GIF, JPEG, WebP, and HEIC ~ispe~ boxes) before falling back to
~new Image()~ on non-iOS runtimes, and always clears ~img.src~ upon completion.

#+begin_src clojurescript :tangle ../../../../src/paradise/shared/utils/helpers.cljs
(defn- u16-be [^js dv off] (.getUint16 dv off false))
(defn- u16-le [^js dv off] (.getUint16 dv off true))
(defn- u32-be [^js dv off] (.getUint32 dv off false))

(defn- read-ascii4 [^js u8 off]
  (when (<= (+ off 4) (.-length u8))
    (str (js/String.fromCharCode (aget u8 off)
                                 (aget u8 (+ off 1))
                                 (aget u8 (+ off 2))
                                 (aget u8 (+ off 3))))))

(defn- parse-png-dims [^js u8 ^js dv]
  (when (and (>= (.-length u8) 24)
             (= (aget u8 0) 0x89)
             (= (read-ascii4 u8 1) "PNG\r"))
    {:width  (u32-be dv 16)
     :height (u32-be dv 20)}))

(defn- parse-gif-dims [^js u8 ^js dv]
  (when (and (>= (.-length u8) 10)
             (let [hdr (read-ascii4 u8 0)]
               (or (= hdr "GIF8"))))
    {:width  (u16-le dv 6)
     :height (u16-le dv 8)}))

(defn- parse-jpeg-dims [^js u8 ^js dv]
  (when (and (>= (.-length u8) 4)
             (= (aget u8 0) 0xFF)
             (= (aget u8 1) 0xD8))
    (let [len (.-length u8)]
      (loop [off 2]
        (when (< (+ off 4) len)
          (if (not= (aget u8 off) 0xFF)
            nil
            (let [marker (aget u8 (inc off))]
              (cond
                (or (= marker 0xFF) (= marker 0x00))
                (recur (inc off))

                (contains? #{0xC0 0xC1 0xC2 0xC3 0xC5 0xC6 0xC7 0xC9 0xCA 0xCB 0xCD 0xCE 0xCF}
                           marker)
                (when (<= (+ off 9) len)
                  {:height (u16-be dv (+ off 5))
                   :width  (u16-be dv (+ off 7))})

                (or (= marker 0xD9) (= marker 0xDA))
                nil

                :else
                (let [seg-len (u16-be dv (+ off 2))]
                  (when (>= seg-len 2)
                    (recur (+ off 2 seg-len))))))))))))

(defn- parse-webp-dims [^js u8 ^js dv]
  (when (and (>= (.-length u8) 30)
             (= (read-ascii4 u8 0) "RIFF")
             (= (read-ascii4 u8 8) "WEBP"))
    (let [chunk (read-ascii4 u8 12)]
      (cond
        (= chunk "VP8X")
        {:width  (inc (bit-or (aget u8 24)
                              (bit-shift-left (aget u8 25) 8)
                              (bit-shift-left (aget u8 26) 16)))
         :height (inc (bit-or (aget u8 27)
                              (bit-shift-left (aget u8 28) 8)
                              (bit-shift-left (aget u8 29) 16)))}

        (= chunk "VP8 ")
        {:width  (bit-and (u16-le dv 26) 0x3FFF)
         :height (bit-and (u16-le dv 28) 0x3FFF)}

        (= chunk "VP8L")
        (let [bits (bit-or (aget u8 21)
                           (bit-shift-left (aget u8 22) 8)
                           (bit-shift-left (aget u8 23) 16)
                           (bit-shift-left (aget u8 24) 24))]
          {:width  (inc (bit-and bits 0x3FFF))
           :height (inc (bit-and (unsigned-bit-shift-right bits 14) 0x3FFF))})

        :else nil))))

(defn- parse-heic-dims [^js u8 ^js dv]
  (when (and (>= (.-length u8) 24)
             (= (read-ascii4 u8 4) "ftyp"))
    (let [len (.-length u8)]
      (loop [i 8]
        (when (<= (+ i 16) len)
          (if (= (read-ascii4 u8 i) "ispe")
            (let [w (u32-be dv (+ i 8))
                  h (u32-be dv (+ i 12))]
              (when (and (pos? w) (pos? h) (<= w 32768) (<= h 32768))
                {:width w :height h}))
            (recur (inc i))))))))

(defn parse-image-dimensions-from-buffer
  "Parses `{:width :height}` from the leading header bytes of PNG/GIF/JPEG/WebP/HEIC."
  [^js ab]
  (when (and ab (pos? (.-byteLength ab)))
    (try
      (let [u8 (js/Uint8Array. ab)
            dv (js/DataView. ab)]
        (or (parse-png-dims u8 dv)
            (parse-jpeg-dims u8 dv)
            (parse-webp-dims u8 dv)
            (parse-gif-dims u8 dv)
            (parse-heic-dims u8 dv)))
      (catch :default _
        nil))))

(defn- probe-image-header-dims [raw-file]
  (try
    (let [slice-fn (.-slice raw-file)
          head     (if (fn? slice-fn)
                     (.slice raw-file 0 (min (or (.-size raw-file) 65536) 65536))
                     raw-file)]
      (-> (file->array-buffer head)
          (p/then parse-image-dimensions-from-buffer)
          (p/catch (fn [_] nil))))
    (catch :default _
      (p/resolved nil))))

(def ^:private max-ios-inline-preview-bytes (* 2 1024 1024))

(defn extract-metadata [raw-file]
  (p/create
   (fn [resolve-fn _]
     (try
       (if-not raw-file
         (do
           (log/warn "Extract-metadata received an empty or undefined file object. Bailing.")
           (resolve-fn nil))
         (let [mime          (or (not-empty (.-type raw-file)) "application/octet-stream")
               filename      (or (not-empty (.-name raw-file)) (str "upload-" (random-uuid)))
               size          (or (.-size raw-file) 0)
               is-image?     (str/starts-with? mime "image/")
               is-video?     (str/starts-with? mime "video/")
               ios?          (platform/ios-runtime?)
               make-preview? (and is-image?
                                  (or (not ios?)
                                      (<= size max-ios-inline-preview-bytes)))]
           (log/info "Safe extract-metadata -> Name:" filename "Mime:" mime "Size:" size)
           (let [fid      (register-file! raw-file)
                 preview  (when make-preview?
                            (try (js/URL.createObjectURL raw-file)
                                 (catch :default e
                                   (log/warn "Failed to create object URL" e)
                                   nil)))
                 base-att {:file-id     fid
                           :mime        mime
                           :mimetype    mime
                           :filename    filename
                           :size        size
                           :preview-url preview}
                 !settled (atom false)
                 finish!  (fn [att]
                            (when (compare-and-set! !settled false true)
                              (resolve-fn att)))]
             (cond
               is-image?
               (-> (probe-image-header-dims raw-file)
                   (p/then
                    (fn [dims]
                      (cond
                        (and dims (pos? (:width dims)) (pos? (:height dims)))
                        (finish! (merge base-att dims))

                        (and (not ios?) preview)
                        (let [img   (js/Image.)
                              timer (js/setTimeout
                                     (fn []
                                       (set! (.-onload img) nil)
                                       (set! (.-onerror img) nil)
                                       (set! (.-src img) "")
                                       (finish! base-att))
                                     2500)
                              done! (fn [att]
                                      (js/clearTimeout timer)
                                      (set! (.-onload img) nil)
                                      (set! (.-onerror img) nil)
                                      (set! (.-src img) "")
                                      (finish! att))]
                          (set! (.-onerror img)
                                (fn [e]
                                  (log/warn "Failed to load image to extract dimensions." e)
                                  (done! base-att)))
                          (set! (.-onload img)
                                (fn []
                                  (done! (assoc base-att
                                                :width  (.-width img)
                                                :height (.-height img)))))
                          (set! (.-src img) preview))

                        :else
                        (finish! base-att))))
                   (p/catch (fn [_] (finish! base-att))))

               (and is-video? (not ios?))
               (let [probe-url (try (js/URL.createObjectURL raw-file)
                                    (catch :default _ nil))]
                 (if-not probe-url
                   (finish! base-att)
                   (let [vid   (js/document.createElement "video")
                         timer (js/setTimeout
                                (fn []
                                  (try
                                    (set! (.-onloadedmetadata vid) nil)
                                    (set! (.-onerror vid) nil)
                                    (.removeAttribute vid "src")
                                    (.load vid)
                                    (catch :default _ nil))
                                  (revoke-object-url! probe-url)
                                  (finish! base-att))
                                2500)
                         done! (fn [att]
                                 (js/clearTimeout timer)
                                 (try
                                   (set! (.-onloadedmetadata vid) nil)
                                   (set! (.-onerror vid) nil)
                                   (.removeAttribute vid "src")
                                   (.load vid)
                                   (catch :default _ nil))
                                 (revoke-object-url! probe-url)
                                 (finish! att))]
                     (set! (.-muted vid) true)
                     (set! (.-playsInline vid) true)
                     (set! (.-preload vid) "metadata")
                     (set! (.-onerror vid)
                           (fn [e]
                             (log/warn "Failed to load video metadata." e)
                             (done! base-att)))
                     (set! (.-onloadedmetadata vid)
                           (fn []
                             (done! (assoc base-att
                                           :width    (.-videoWidth vid)
                                           :height   (.-videoHeight vid)
                                           :duration (when (js/Number.isFinite (.-duration vid))
                                                       (Math/round (* 1000 (.-duration vid))))))))
                     (set! (.-src vid) probe-url))))

               :else
               (finish! base-att)))))
       (catch :default e
         (log/error "Catastrophic error in extract-metadata:" e)
         (resolve-fn nil))))))

(defn normalize-keyword
  "Converts a non-empty keyword, string, or symbol into a keyword (preserving namespace); returns nil otherwise."
  [x]
  (cond
    (keyword? x) x
    (symbol? x)  (keyword x)
    (string? x)  (let [s (str/replace x #"^:" "")]
                   (when (seq s) (keyword s)))
    :else        nil))

  "Strips trailing LiveKit/device suffixes (e.g. `|device`) from a matrix-style user ID."
  [raw-id]
  (when (and raw-id (not (str/blank? (str raw-id))))
    (or (re-find #"^@[^:]+:[^:|]+" (str raw-id))
        (str raw-id))))

(defn deep-merge
  "Recursively merges maps, ignoring nil inputs."
  [& maps]
  (apply merge-with

           (if (and (map? a) (map? b))
             (deep-merge a b)
             b))
         (remove nil? maps)))

(defn resolve-item-label
  "Resolves a declarative item's display label from `:title-key`/`:label-key` or literal `:title`/`:label`."
  [tr {:keys [title title-key title-args label label-key label-args]}]
  (let [k    (normalize-keyword (or title-key label-key))
        args (or title-args label-args)]
    (cond
      k     (if (seq args) (tr [k] (vec args)) (tr [k]))
      title title
      label label
      :else "")))

(defn resolve-item-desc
  "Resolves a declarative item's description/subtitle from `:desc-key`/`:subtitle-key` or literal `:desc`/`:subtitle`."
  [tr {:keys [desc desc-key desc-args subtitle subtitle-key subtitle-args]}]
  (let [k    (normalize-keyword (or desc-key subtitle-key))
        args (or desc-args subtitle-args)]
    (cond
      k        (if (seq args) (tr [k] (vec args)) (tr [k]))
      desc     desc
      subtitle subtitle
      :else    "")))

(defn resolve-dm-display
  "Derives DM / Group-DM peer list, avatar, and display name from room membership."
  [{:keys [is-dm? is-group-dm? members-map my-id avatar-url name only-replace-loading?]}]
  (let [other-users (when (and is-dm? (seq members-map))
                      (->> (vals members-map)
                           (remove #(= (:user-id %) my-id))))
        group-dm?   (boolean (or is-group-dm? (> (count other-users) 1)))
        other-user  (when (and is-dm? (not group-dm?))
                      (first other-users))
        final-av    (if group-dm?
                      avatar-url
                      (or avatar-url (:avatar-url other-user)))
        placeholder (or (str/blank? name) (= name "Loading..."))
        final-name  (cond
                      (and group-dm? placeholder (seq other-users))
                      (str/join ", " (map :display-name other-users))

                      (and is-dm? other-user (or (not only-replace-loading?) placeholder))
                      (:display-name other-user)

                      :else
                      name)]
    {:other-users  other-users
     :group-dm?    group-dm?
     :other-user   other-user
     :avatar-url   final-av
     :display-name final-name}))

(defn capacitor-platform
  "Returns the active Capacitor platform string (e.g. `\"ios\"`, `\"android\"`, `\"web\"`), or nil."
  []
  (try
    (when (exists? js/window)
      (let [^js cap (aget js/window "Capacitor")]
        (when (and (some? cap) (fn? (.-getPlatform cap)))
          (str (.getPlatform cap)))))
    (catch :default _
      nil)))

(defn capacitor-mobile?
  "Returns true when running inside a native iOS or Android Capacitor shell."
  []
  (contains? #{"ios" "android"} (capacitor-platform)))

(defn ios-runtime?
  "Returns true on native iOS Capacitor shells as well as iOS/iPadOS Safari/WebKit."
  []
  (try
    (or (= "ios" (capacitor-platform))
        (when (exists? js/navigator)
          (let [ua  (str (or (.-userAgent js/navigator) ""))
                plt (str (or (.-platform js/navigator) ""))
                mtp (js/Number (or (.-maxTouchPoints js/navigator) 0))]
            (or (boolean (re-find #"(?i)iPhone|iPad|iPod" ua))
                (and (= plt "MacIntel") (> mtp 1))))))
    (catch :default _
      false)))

(defn mobile-runtime?
  "Returns true on native mobile shells (iOS/Android) or mobile browsers."
  []
  (try
    (or (capacitor-mobile?)
        (ios-runtime?)
        (when (exists? js/navigator)
          (boolean (re-find #"(?i)Android" (str (or (.-userAgent js/navigator) ""))))))
    (catch :default _
      false)))

(defn parse-edn-map
  "Parses an EDN string into a map (or passes an existing map through), returning `{}` on failure."
  [raw-val]
  (cond
    (map? raw-val)    raw-val
    (string? raw-val) (try

(defn decode-edn-field
  "Decodes an EDN-encoded string field (e.g. :options-edn) from a worker payload,
  falling back to `fallback-key` when absent or unreadable."
  ([m edn-key]
   (decode-edn-field m edn-key nil))
  ([m edn-key fallback-key]
   (or (when-let [s (get m edn-key)]
         (when (string? s)
           (try
             (reader/read-string s)
             (catch :default e
               (log/warn "Failed to parse" edn-key ":" e)
               nil))))
       (when fallback-key
         (get m fallback-key)))))
