
** Media Component

**** Preface

This is should enable handling media in a more idiomatic manner across both
offloaded rendering and regular rendering. Both should able to use this
method more or less the same without consideration with how it might behave.

(defn- resolve-active-user-id [props]
  (or (:user-id props)
      (:userId props)
      (:active-user-id @rf-db/app-db)))

(defn- extract-cache-key [user-id uri source-map width height]
  (let [base-key (or (:uri source-map)
                     (get-in source-map [:file :url])
                     (when (and source-map (not (map? source-map)))
                       (or (.-uri source-map)
                           (some-> (.-file source-map) .-url)
                           (some-> (.-plain source-map) .-url)))
                     uri)
        sized    (if (and base-key width height)
                   (str base-key "?w=" width "&h=" height)
                   base-key)]
    (when sized
      (str (or user-id "anon") "::" sized))))

(defn- hide-spinner! [node]
  (when node
    (when-let [prev (.-previousElementSibling node)]
      (when (and (.-classList prev)
                 (.contains (.-classList prev) "message-image-placeholder")
                 (.-style prev))
        (set! (.-display (.-style prev)) "none")))))

(defn- notify-error! [node]
  (when node
    (when-let [err-fn (.-__paradiseOnError node)]
      (err-fn))))

#+end_src



While building out the virtualizer I began to realize the lack of uniformity in how we rendered
components still.

While mxc-image solves the problem in a blunt sort of manner... it would be more sensible that
it not define the tag it would be using. That way we can allow ANY mxc URL to have deferred
evaluation.

In a Matrix client that would be every URL that could be rendered on the screen.

Hence, any component defining a src block would be utilizing an mxc. It would then make more
sense to remove the burden of evaluation from the developer and instead treat components
in a natural form.

Now, a problem we encountered often was that the Hiccup compiler adds an annoying amount
of overhead that isn't necessary. Most of the time it is completely acceptable, but for
extremely performance driven components it is a liability.

It may then be reasonable to assume that we might find ourselves crafting the 'mxc-image'
as a raw React component.


**** Media Fetch

#+begin_src clojurescript :tangle ../../../src/paradise/media/component.cljs
(defonce !native-media-cache (js/Map.))

(defn clear-media-cache! []
  (.forEach !native-media-cache
            (fn [entry _]
              (when-let [url (.-url entry)]
                (when (str/starts-with? (str url) "blob:")
                  (js/URL.revokeObjectURL url)))))
  (.clear !native-media-cache))

(defn- resolve-pending-entry! [cache-key status url fallback-url]
  (when-let [entry (.get !native-media-cache cache-key)]
    (let [nodes (.-nodes entry)]
      (set! (.-status entry) status)
      (set! (.-url entry) url)
      (when nodes
        (.forEach nodes
                  (fn [node]
                    (if url
                      (set! (.-src node) url)
                      (when (= status "error")
                        (notify-error! node)))
                    (when (= status "ready")
                      (hide-spinner! node))))
        (.clear nodes))
      (when (<= (.-refs entry) 0)
        (when (and url (str/starts-with? (str url) "blob:"))
          (js/URL.revokeObjectURL url))
        (.delete !native-media-cache cache-key))
      (when (and (= status "error") (not url))
        (.delete !native-media-cache cache-key)))))

(defn- schedule-cache-cleanup! [cache-key]
  (js/setTimeout
   (fn []
     (when-let [entry (.get !native-media-cache cache-key)]
       (when (and (<= (.-refs entry) 0)
                  (not= (.-status entry) "pending"))
         (when-let [url (.-url entry)]
           (when (str/starts-with? (str url) "blob:")
             (js/URL.revokeObjectURL url)))
         (.delete !native-media-cache cache-key))))
   100))

(defn execute-media-fetch! [cache-key user-id uri source-map mime-type fallback-url room-id event-id width height method]
  (go
    (try
      (let [res (<! (mesh/do-with-thread! :media-pool
                                          {:handler   :get-media
                                           :arguments {:user-id    user-id
                                                       :room-id    room-id
                                                       :event-id   event-id
                                                       :uri        uri
                                                       :source     uri
                                                       :source-map source-map
                                                       :width      width
                                                       :height     height
                                                       :method     method}}))]
        (case (:status res)
          ("unencrypted" "direct")
          (resolve-pending-entry! cache-key "ready" (:url res) fallback-url)

          "success"
          (if-let [entry (.get !native-media-cache cache-key)]
            (if (<= (.-refs entry) 0)
              (.delete !native-media-cache cache-key)
              (let [buffer (:bytes res)
                    blob   (js/Blob. #js [buffer] #js {:type mime-type})
                    url    (js/URL.createObjectURL blob)]
                (resolve-pending-entry! cache-key "ready" url fallback-url)))
            nil)

          "error"
          (do
            (js/console.error "Media load failed:" (:msg res))
            (resolve-pending-entry! cache-key "error" fallback-url fallback-url))

          (do
            (js/console.error "Unknown status from media worker:" res)
            (resolve-pending-entry! cache-key "error" fallback-url fallback-url))))
      (catch :default e
        (js/console.error "Media fetch go-block error:" e)
        (resolve-pending-entry! cache-key "error" fallback-url fallback-url)))))
(defn- retain-media! [node cache-key user-id uri source-map mime-type fallback-url room-id event-id width height method]
  (if-let [entry (.get !native-media-cache cache-key)]
    (do
      (set! (.-refs entry) (+ (.-refs entry) 1))
      (case (.-status entry)
        "ready"
        (do
          (set! (.-src node) (.-url entry))
          (hide-spinner! node))

        "pending"
        (.add (.-nodes entry) node)

        "error"
        (if fallback-url
          (set! (.-src node) fallback-url)
          (notify-error! node))

        nil))
    (let [nodes (js/Set.)]
      (.add nodes node)
      (.set !native-media-cache
            cache-key
            #js {:url    nil
                 :status "pending"
                 :refs   1
                 :nodes  nodes})
      (execute-media-fetch! cache-key user-id uri source-map mime-type fallback-url room-id event-id width height method))))

(defn- release-media! [node cache-key]
  (when-let [entry (.get !native-media-cache cache-key)]
    (when-let [nodes (.-nodes entry)]
      (.delete nodes node))
    (set! (.-refs entry) (- (.-refs entry) 1))
    (when (<= (.-refs entry) 0)
      (schedule-cache-cleanup! cache-key))))

**** Media Native Component
This is the component generated under the hood for deferred rendering.
We make sure we don't burden the main thread with evaluating components
as much as possible. Instead we just want to 'pop' the image in.

#+begin_src clojurescript :tangle ../../../src/paradise/media/component.cljs
(defn create-media-ref [props]
  (let [user-id       (resolve-active-user-id props)
        uri           (:uri props)
        source-map    (or (:source-map props) (:sourceMap props))
        mime-type     (or (:mime-type props) (:mimeType props) "image/png")
        fallback-url  (or (:fallback-url props) (:fallbackUrl props))
        on-error      (or (:on-error props) (:onError props))
        room-id       (or (:room-id props) (:roomId props))
        event-id      (or (:event-id props) (:eventId props))
        width         (:width props)
        height        (:height props)
        method        (:method props)
        cache-key     (extract-cache-key user-id uri source-map width height)
        !mounted-node (atom nil)]
    (fn [node]
      (if node
        (do
          (set! (.-__paradiseOnError node) on-error)
          (when (and cache-key (not= @!mounted-node node))
            (when-let [prev @!mounted-node]
              (release-media! prev cache-key))
            (reset! !mounted-node node)
            (retain-media! node cache-key user-id uri source-map mime-type fallback-url room-id event-id width height method)))
        (when-let [prev @!mounted-node]
          (set! (.-__paradiseOnError prev) nil)
          (reset! !mounted-node nil)
          (when cache-key
            (release-media! prev cache-key)))))))

(re-frame/reg-event-fx
 :media/download
 (fn [{:keys [db]} [_ {:keys [user-id uri source-map filename mime-type room-id event-id]}]]
   (let [effective-uid (or user-id (:active-user-id db))]
     (go
       (try
         (let [res (<! (mesh/do-with-thread! :media-pool
                                             {:handler   :get-media
                                              :arguments {:user-id    effective-uid
                                                          :room-id    room-id
                                                          :event-id   event-id
                                                          :uri        uri
                                                          :source     uri
                                                          :source-map source-map
                                                          :download?  true}}))]
           (case (:status res)
             "success"
             (let [blob (js/Blob. #js [(:bytes res)] #js {:type (or mime-type "application/octet-stream")})
                   url  (js/URL.createObjectURL blob)
                   link (js/document.createElement "a")]
               (set! (.-href link) url)
               (set! (.-download link) (or filename "download"))
               (set! (.-display (.-style link)) "none")
               (js/document.body.appendChild link)
               (.click link)
               (js/document.body.removeChild link)
               (js/setTimeout #(js/URL.revokeObjectURL url) 10000))

             ("direct" "unencrypted")
             (when-let [direct-url (:url res)]
               (let [link (js/document.createElement "a")]
                 (set! (.-href link) direct-url)
                 (set! (.-download link) (or filename "download"))
                 (set! (.-target link) "_blank")
                 (set! (.-rel link) "noopener noreferrer")
                 (set! (.-display (.-style link)) "none")
                 (js/document.body.appendChild link)
                 (.click link)
                 (js/document.body.removeChild link)))

             (js/console.error "File download failed:" (:msg res))))
         (catch :default e
           (js/console.error "File download error:" e)))))
   {}))

(defn ^:react media-native [props]
  (let [tag-type     (or (:tag-type props) (:tagType props) "img")
        class        (or (:class props) (:className props) (:class-name props))
        style        (:style props)
        alt          (:alt props)
        fallback-url (or (:fallback-url props) (:fallbackUrl props))
        on-error     (or (:on-error props) (:onError props))
        on-click     (or (:on-click props) (:onClick props))
        controls     (:controls props)
        handle-err   (fn [e]
                       (let [el (.-currentTarget e)]
                         (if (and fallback-url el (not= (.-src el) fallback-url))
                           (set! (.-src el) fallback-url)
                           (when on-error
                             (on-error e)))))]
    (react/createElement tag-type
                         #js {:className      class
                              :style          (clj->js style)
                              :alt            alt
                              :controls       controls
                              :onClick        on-click
                              :referrerPolicy (when (= tag-type "img") "no-referrer")
                              :loading        (when (= tag-type "img") "lazy")
                              :decoding       (when (= tag-type "img") "async")
                              :onError        handle-err
                              :ref            (create-media-ref props)})))
